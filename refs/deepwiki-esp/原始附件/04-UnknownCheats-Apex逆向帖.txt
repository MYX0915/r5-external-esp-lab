void GlowFeature::InitializeGlowSettings() {
    if (initialized || !g_Driver || !g_ApexBase) return;
    g_Driver->ForceFlush();
    uintptr_t highlightSettings = Read<uintptr_t>(g_ApexBase + OFF_GLOW_HIGHLIGHTSettings);
    if (!highlightSettings) return;
 
    HighlightFunctionBits functionBits;
    functionBits.functionBits[0] = 2;
    functionBits.functionBits[1] = 125;
    functionBits.functionBits[2] = 64;
    functionBits.functionBits[3] = 64; // 渲染方式相同
 
    // 可见索引 (70)
    HighlightParameter visibleParam;
    for (int i = 0; i < 4; ++i) visibleParam.parameter[i] = GlowVisible[i];
    Write<HighlightFunctionBits>(highlightSettings + OFF_HIGHLIGHT_TYPE_SIZE * visibleIndex + 0, functionBits);
    Write<HighlightParameter>(highlightSettings + OFF_HIGHLIGHT_TYPE_SIZE * visibleIndex + 4, visibleParam);
 
    // 不可见索引 (80)
    HighlightParameter invisibleParam;
    for (int i = 0; i < 4; ++i) invisibleParam.parameter[i] = GlowInvisible[i];
    Write<HighlightFunctionBits>(highlightSettings + OFF_HIGHLIGHT_TYPE_SIZE * invisibleIndex + 0, functionBits);
    Write<HighlightParameter>(highlightSettings + OFF_HIGHLIGHT_TYPE_SIZE * invisibleIndex + 4, invisibleParam);
 
    initialized = true;
}
 
void GlowFeature::SetEntityGlow(const Entity& entity, bool isVisible) {
    if (!entity.address) return;
 
    int targetIndex = isVisible ? visibleIndex : invisibleIndex;
 
    auto it = entityGlowState.find(entity.address);
    if (it != entityGlowState.end() && it->second == targetIndex) {
        return; // 状态未变
    }
 
    Write<int>(entity.address + OFF_GLOW_THROUGH_WALL, 1);
    Write<int>(entity.address + OFF_GLOW_FIX, 2);
    Write<float>(entity.address + OFF_GLOW_DISTANCE, 99999.0f);
    Write<int>(entity.address + OFF_GLOW_ENABLE, 1);
    Write<int>(entity.address + OFF_GLOW_HIGHLIGHT_ID + 0, targetIndex);
 
    entityGlowState[entity.address] = targetIndex;
}
 
void GlowFeature::DisableEntityGlow(uintptr_t entityAddress) {
    if (!entityAddress) return;
    Write<int>(entityAddress + OFF_GLOW_HIGHLIGHT_ID + 0, 0);
    Write<int>(entityAddress + OFF_GLOW_THROUGH_WALL, 0);
    Write<int>(entityAddress + OFF_GLOW_FIX, 0);
    Write<float>(entityAddress + OFF_GLOW_DISTANCE, 0.0f);
    Write<int>(entityAddress + OFF_GLOW_ENABLE, 0);
}
 
void GlowFeature::SetEnabled(bool enabled) {
    bool oldState = g_GlowEnabled;
    g_GlowEnabled = enabled;
 
    if (oldState && !enabled) {
        for (const auto& pair : entityGlowState) {
            DisableEntityGlow(pair.first);
        }
        entityGlowState.clear();
    }
}

    if (!IsEnabled()) return;
 
    DWORD now = GetTickCount();
    if (now - lastUpdateTime < UPDATE_INTERVAL) return;
    lastUpdateTime = now;
 
    if (!initialized) {
        InitializeGlowSettings();
    }
 
    std::vector<Entity> entities;
